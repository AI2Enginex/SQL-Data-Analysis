-- Year-to-Date (YTD) Sales
-- This calculates the total sales from the beginning of the year to the current date.
YTD Sales = TOTALYTD(SUM(SalesData[Sales Amount]), DateTable[Date])

-- Alternative using DATESYTD
YTD Sales DATES = CALCULATE(SUM(SalesData[Sales Amount]), DATESYTD(DateTable[Date]))

-- Example:
-- Given Sales Data:
-- 2024-01-05: 500
-- 2024-02-10: 700
-- 2024-03-20: 600
-- 2024-06-15: 900
-- If today is June 15, 2024, YTD Sales will be 500 + 700 + 600 + 900 = 2700

-- Quarter-to-Date (QTD) Sales
-- This calculates the total sales from the beginning of the current quarter to the current date.
QTD Sales = TOTALQTD(SUM(SalesData[Sales Amount]), DateTable[Date])

-- Alternative using DATESQTD
QTD Sales DATES = CALCULATE(SUM(SalesData[Sales Amount]), DATESQTD(DateTable[Date]))

-- Example:
-- Given Sales Data:
-- 2024-04-05: 300
-- 2024-05-10: 600
-- 2024-06-15: 900
-- If today is June 15, 2024 (Q2), QTD Sales will be 300 + 600 + 900 = 1800

-- Month-to-Date (MTD) Sales
-- This calculates the total sales from the beginning of the current month to the current date.
MTD Sales = TOTALMTD(SUM(SalesData[Sales Amount]), DateTable[Date])

-- Alternative using DATESMTD
MTD Sales DATES = CALCULATE(SUM(SalesData[Sales Amount]), DATESMTD(DateTable[Date]))

-- Example:
-- Given Sales Data:
-- 2024-06-05: 300
-- 2024-06-10: 400
-- 2024-06-15: 500
-- If today is June 15, 2024, MTD Sales will be 300 + 400 + 500 = 1200

-- MTD Sales (Weekdays Only)
-- This calculates month-to-date sales but excludes weekends.
MTD Sales Weekdays = 
CALCULATE(
    SUM(SalesData[Sales Amount]), 
    DATESMTD(DateTable[Date]),
    NOT WEEKDAY(DateTable[Date], 1) IN {1, 7} -- Excludes Saturdays (7) and Sundays (1)
)

-- Example:
-- Given Sales Data:
-- 2024-06-01 (Sat): 100 (excluded)
-- 2024-06-03 (Mon): 200
-- 2024-06-05 (Wed): 300
-- 2024-06-08 (Sat): 400 (excluded)
-- 2024-06-10 (Mon): 500
-- If today is June 10, 2024, MTD Sales (Weekdays Only) will be 200 + 300 + 500 = 1000

-- Total Year-to-Date (YTD) Sales
-- This calculates the total YTD sales across all dates, ignoring filters.
Total YTD Sales = TOTALYTD(SUM(SalesData[Sales Amount]), DateTable[Date], ALL(DateTable))

-- Total Quarter-to-Date (QTD) Sales
-- This calculates the total QTD sales across all dates, ignoring filters.
Total QTD Sales = TOTALQTD(SUM(SalesData[Sales Amount]), DateTable[Date], ALL(DateTable))

-- Total Month-to-Date (MTD) Sales
-- This calculates the total MTD sales across all dates, ignoring filters.
Total MTD Sales = TOTALMTD(SUM(SalesData[Sales Amount]), DateTable[Date], ALL(DateTable))

-- Same Period Last Year (YoY)
-- This calculates sales for the same period in the previous year.
Sales LY = CALCULATE(SUM(SalesData[Sales Amount]), SAMEPERIODLASTYEAR(DateTable[Date]))

-- Previous Month Sales
-- This calculates sales for the previous month.
Previous Month Sales = CALCULATE(SUM(SalesData[Sales Amount]), PREVIOUSMONTH(DateTable[Date]))

-- Previous Quarter Sales
-- This calculates sales for the previous quarter.
Previous Quarter Sales = CALCULATE(SUM(SalesData[Sales Amount]), PREVIOUSQUARTER(DateTable[Date]))

-- Previous Year Sales
-- This calculates sales for the previous year.
Previous Year Sales = CALCULATE(SUM(SalesData[Sales Amount]), PREVIOUSYEAR(DateTable[Date]))

-- Running Total Over All Time
-- This calculates a cumulative sum of sales over all time up to the current date.
Cumulative Sales = CALCULATE(SUM(SalesData[Sales Amount]), FILTER(ALL(DateTable), DateTable[Date] <= MAX(DateTable[Date])))

-- Rolling 12-Month Sales
-- This calculates total sales for the last 12 months from the current date.
Rolling 12M Sales = CALCULATE(SUM(SalesData[Sales Amount]), DATESINPERIOD(DateTable[Date], MAX(DateTable[Date]), -12, MONTH))

-- Parallel Period: Sales for the Same Month in the Previous Year
-- This calculates sales for the same month in the previous year using PARALLELPERIOD.
Parallel Period Sales = CALCULATE(SUM(SalesData[Sales Amount]), PARALLELPERIOD(DateTable[Date], -1, YEAR))

-- Year-over-Year Growth
-- This calculates the percentage growth of YTD sales compared to the same period last year.
YoY Growth = DIVIDE([YTD Sales] - [Sales LY], [Sales LY], 0)

-- Quarter-over-Quarter Growth
-- This calculates the percentage growth of QTD sales compared to the previous quarter.
QoQ Growth = DIVIDE([QTD Sales] - [Previous Quarter Sales], [Previous Quarter Sales], 0)

-- Month-over-Month Growth
-- This calculates the percentage growth of MTD sales compared to the previous month.
MoM Growth = DIVIDE([MTD Sales] - [Previous Month Sales], [Previous Month Sales], 0)

-- Sales for Last 30 Days
-- This calculates total sales for the last 30 days from the current date.
Last 30 Days Sales = CALCULATE(SUM(SalesData[Sales Amount]), DATESINPERIOD(DateTable[Date], MAX(DateTable[Date]), -30, DAY))
